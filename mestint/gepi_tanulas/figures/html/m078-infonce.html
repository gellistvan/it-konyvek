def info_nce_loss(z, temperature=0.5):    N = z.shape[0] // 2    sim = (z @ z.T) / temperature                      # [2N, 2N]    mask = torch.eye(2 * N, device=z.device, dtype=torch.bool)    sim = sim.masked_fill(mask, float("-inf"))        # saját index kizárása    labels = torch.arange(2 * N, device=z.device)    labels = torch.where(labels < N, labels + N, labels - N)   # i -> i+N, i+N -> i    loss = F.cross_entropy(sim, labels)                # −log softmax a pozitív oszlopon, átlag    return loss

Számított, véletlen vektorokkal (seed 4; a második nézet az első zajos, normált változata), nem mérés. Az „Azonos vektorok” gomb egyenletes logitokat ad: ekkor L = ln 7 ≈ 1,95 (N = 4; N = 32-nél ln 63 ≈ 4,14).
