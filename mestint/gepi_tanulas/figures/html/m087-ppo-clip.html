def ppo_policy_loss(new_log_prob, old_log_prob, advantage, mask, eps=0.2):    ratio = torch.exp(new_log_prob - old_log_prob)           # [B, L]    clipped = torch.clamp(ratio, 1.0 - eps, 1.0 + eps)       # [B, L]    per_token = torch.minimum(ratio * advantage, clipped * advantage)    return -(per_token * mask).sum() / mask.sum()            # skalár
