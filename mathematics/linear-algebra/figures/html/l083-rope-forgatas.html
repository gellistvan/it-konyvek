def rotate_2d_pairs(x, angle_rates):    # x: (d,) vektor; angle_rates: (d/2,) szögek    x_pairs = x.reshape(-1, 2)    cos = np.cos(angle_rates)    sin = np.sin(angle_rates)    rotated = np.stack([        x_pairs[:, 0] * cos - x_pairs[:, 1] * sin,        x_pairs[:, 0] * sin + x_pairs[:, 1] * cos    ], axis=1)    return rotated.reshape(-1)q_m = rotate_2d_pairs(u, m * theta)k_n = rotate_2d_pairs(v, n * theta)dot_rotated = q_m @ k_nrelative = u @ rotate_2d_pairs(v, (n - m) * theta)
