def softmax_stabil(s):    s_shifted = s - np.max(s)       # a maximum kivonása: az eredmény nem változik    exp_s = np.exp(s_shifted)    return exp_s / np.sum(exp_s)probs = softmax_stabil((logits + eltolás) / tau)   # τ = 1: a könyv példája
